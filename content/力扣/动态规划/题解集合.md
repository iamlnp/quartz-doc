# 例题

## 最长回文字符串
```c++
class Solution {
public:

    string longestPalindrome(string s) {
            //使用二维数组[i,j]表示左右边界
            //双层遍历：外层 - 字符串长度； 内层 - 枚举左边界，通过长度可以算出右边界
            //[i,j]回文，那么 [i + 1, j - 1]是回文且Vi = Vj
            //如果[i + 1, j - 1]非回文，那么[i, j]非回文
            //如果字符串长度是1，那么回文
            //如果字符串长度是2，那么只有Vi = Vj时为回文

            if (s.size() < 2) {
                return s;
            }

            int len = s.size();
            //使用构造函数初始化vector,形成一个二维数组
            vector<vector<int>> dp(len, vector<int>(len));

            int maxLen = 1;
            int begin = 0;
            

            for (int i = 0; i < len; i++) {
                dp[i][i] = 1;  //使用二维数组表示范围，[i,i]表示长度为1，那么必定是回文
            }

            for (int l = 1; l < len; l++) {
                for (int left = 0; left < len; left++) {
                    //获取右侧边界
                    int right = left + l;
                    if (right >= len) right = len - 1;
                    
                    if (s[left] == s[right] && dp[left + 1][right - 1]) {
                        dp[left][right] = 1;
                        if (right - left > maxLen) { 
                            maxLen = right - left;
                            begin = left;
                        } 
                    } else {
                        dp[left][right] = dp[left + 1][right - 1];
                    }
                }
            }
            
            return s.substr(begin, maxLen);
        }
};
```
