use std::cmp::min;
impl Solution {
    pub fn min_sub_array_len(target: i32, nums: Vec<i32>) -> i32 {
        let mut i:i32 = 0;
        let mut j:i32 = 0;
        let mut sum:i32 = 0;
        let mut min_len = i32::MAX;
        let mut sub_len:i32 = 0;
        while j < nums.len().try_into().unwrap() {
            sum += nums[j as usize];
            while sum >= target {
                sub_len = j - i + 1;
                min_len = min(sub_len, min_len);
                sum -= nums[i as usize];
                i += 1;
            }
            j += 1;
        }
        if min_len == i32::MAX {
            0
        } else {
            min_len
        }
    }
}
