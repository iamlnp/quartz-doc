假设数组的数量是N，数组的数值范围是K（即数组中max-min的范围），如果K远小于N

```c++
void count_sort(vector<int> &vec) {
    //遍历数组，找到每个元素的计数
    std::map<int, int> count;
    for (int i = 0; i < vec.size(); i++) {
        int key = vec[i];

        if (count.count(key)) {
            count[key]++;
        } else {
            count.insert(make_pair(key, 1));
        }
    }

    //根据计数，重排数据
    int pos = 0;
    for (auto it = count.begin(); it != count.end(); it++) {
        while (it->second > 0) {
            vec[pos] = it->first;
            it->second--;
        }
    }
}
```
